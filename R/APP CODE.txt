#Locate each file, in spite of its casing
find_data <- function(f) {
  p <- file.path(data_dir, f)
  if (file.exists(p)) return(p)
  all <- list.files(data_dir, full.names = TRUE)
  hit <- all[tolower(basename(all)) == tolower(f)]
  if (length(hit) == 0) NULL else hit[1]
}

#The pre-calculated grid is now loaded and built on-demand, with respect to what the user chooses.
make_template <- function(res) {
  terra::rast(
    xmin = min(sapply(risk_bb, function(b) b[["xmin"]])),
    xmax = max(sapply(risk_bb, function(b) b[["xmax"]])),
    ymin = min(sapply(risk_bb, function(b) b[["ymin"]])),
    ymax = max(sapply(risk_bb, function(b) b[["ymax"]])),
    resolution = res, crs = "EPSG:32643")
}

#The coarsest cell size is selected as automatic, such as at 500 metres, or 1000 metres
pick_res <- function(dist_m, choice) {
  if (!identical(choice, "auto")) return(as.numeric(choice))
  ok <- c(500, 250, 100, 50)
  ok <- ok[ok <= dist_m / 2]
  if (length(ok) == 0) 50 else max(ok)
}

# For IDW squared, the if a grid is missing, a run of this happens once
idw2_score <- function(dist_r, D, d0) {
  d_eff <- terra::ifel(dist_r < d0, d0, dist_r)
  terra::ifel(dist_r <= D, (1 / d_eff^2 - 1 / D^2) / (1 / d0^2 - 1 / D^2), 0)
}

#Quartic KDE is loaded here
kde_quartic <- function(counts, h, res) {
  k <- ceiling(h / res)
  ax <- (-k:k) * res
  d  <- sqrt(outer(ax^2, ax^2, "+"))
  w  <- ifelse(d < h, (3 / pi) * (1 - (d / h)^2)^2 / h^2 * 1e6, 0)
  terra::focal(counts, w = w, fun = "sum", na.rm = TRUE)
}

#One image or map is printed, with one set of villages per region, to prevent multiple village imagery being sent through to be processed by the app
village_raster <- function(v, values, max_cells = 2e6) {
  vv <- terra::vect(sf::st_transform(v, 32643))
  vv$val <- values
  ev  <- as.vector(terra::ext(vv))
  res <- max(200, ceiling(sqrt((ev[["xmax"]] - ev[["xmin"]]) * (ev[["ymax"]] - ev[["ymin"]]) / max_cells) / 50) * 50)
  tpl <- terra::rast(terra::ext(vv), resolution = res, crs = "EPSG:32643")
  terra::project(terra::rasterize(vv, tpl, field = "val"), "EPSG:3857", method = "near")
}
img_bytes <- 16 * 1024 * 1024   # allow a larger image before leaflet coarsens it

#The raster here is reduced in size to the maximum size it can produce an output and maintain accuracy
display_max_cells <- 4.5e6
display_raster <- function(r, max_cells = display_max_cells) {
  fact <- ceiling(sqrt(terra::ncell(r) / max_cells))
  if (fact > 1) r <- terra::aggregate(r, fact = fact, fun = "modal", na.rm = TRUE)
  terra::project(r, "EPSG:3857", method = "near")
}

#Here, for the Fast maps, the village lines are drawn using single lines as opposed to many, smaller connecting lines to ensure that
#a broad summary of the villages can be viewed for the four regions without the computational load
line_cache <- new.env()
village_lines <- function(regs) {
  do.call(rbind, lapply(regs, function(r) {
    if (is.null(line_cache[[r]])) {
      b <- sf::st_boundary(sf::st_geometry(Villages_disp[Villages_disp$region == r, ]))
      parts <- unlist(lapply(b, function(x) {
        if (inherits(x, "MULTILINESTRING")) unclass(x) else list(unclass(x))
      }), recursive = FALSE)
      line_cache[[r]] <- sf::st_sf(region = r,
                                   geometry = sf::st_sfc(sf::st_multilinestring(parts), crs = 4326))
    }
    line_cache[[r]]
  }))
}

#Below is the base code for the banner, including rotations
fact_banner <- function(facts, seconds) {
  facts <- vapply(as.character(facts), fmt_text, character(1), USE.NAMES = FALSE)  
  tagList(
    div(class = "fact-banner",
        div(class = "fact-label", "\U0001F3D4 Did you know?"),
        div(id = "fact-text", HTML(facts[1])),
        div(class = "fact-dots", id = "fact-dots")),
    tags$script(HTML(sprintf("
      $(function() {
        var facts = %s, secs = %d, i = 0;
        var el = $('#fact-text'), dots = $('#fact-dots');
        if (!facts.length) return;
        facts.forEach(function(f, k) {
          dots.append('<span class=\"fact-dot' + (k === 0 ? ' active' : '') + '\"></span>');
        });
        el.html(facts[0]);
        if (facts.length < 2) return;
        setInterval(function() {
          el.fadeOut(600, function() {
            i = (i + 1) %% facts.length;
            el.html(facts[i]).fadeIn(600);
            dots.children().removeClass('active').eq(i).addClass('active');
          });
        }, secs * 1000);
      });",
      as.character(jsonlite::toJSON(unname(as.character(facts)))),
      as.integer(seconds))))
  )
}

#For text formatting, et al. is made italics and the YouTube links are made clickable
linkify <- function(x) {
  pat <- "(?<![\"'>=])https?://[^\\s<>\"']+"
  m   <- gregexpr(pat, x, perl = TRUE)
  regmatches(x, m) <- lapply(regmatches(x, m), function(u) {
    vapply(u, function(one) {
      core  <- sub("[.,;:)]+$", "", one)                \
      trail <- substr(one, nchar(core) + 1, nchar(one))
      url   <- gsub("&", "&amp;", gsub("&amp;", "&", core, fixed = TRUE), fixed = TRUE)
      paste0("<a href=\"", url, "\" target=\"_blank\" rel=\"noopener noreferrer\">", url, "</a>", trail)
    }, character(1), USE.NAMES = FALSE)
  })
  x
}

#Kilometres squared is rewritten with the subscript
sup_km2 <- function(x) gsub("km2(?![A-Za-z0-9])", "km<sup>2</sup>", x, perl = TRUE)

fmt_text <- function(x) linkify(sup_km2(italic_etal(x)))

#Keeps the block of text as paragraphs
para <- function(x) {
  if (is.null(x) || length(x) == 0 || is.na(x[1])) return(NULL)   # a missing entry shows nothing instead of crashing
  parts <- trimws(strsplit(x, "\n\\s*\n")[[1]])
  tagList(lapply(parts[nzchar(parts)], function(t) p(HTML(gsub("\n", "<br>", fmt_text(t))))))
}

#For the statistical methods tab, the dropdown arrows are kept permanently open
static_panel <- function(title, ...) {
  div(class = "panel panel-default",
      div(class = "panel-heading", h4(class = "panel-title", title)),
      div(class = "panel-body", ...))
}

#The description of the methods are written on the left while the formulae are displayed on the right
method_row <- function(title, text, formula) {
  fluidRow(class = "method-row",
           column(width = 6, static_panel(title, para(text))),
           column(width = 6, class = "split-right",
                  static_panel(paste0(title, ": formulae"), para(formula))))
}

  # Grid helpers are utilised here to extract the grids, and fall back on recalculation if it is not found (but this is an absolute backup)

  get_template <- function(res) {
    k <- paste0("tpl_", res)

    }

  get_slope_grid <- function(res) {
    k <- paste0("slope_", res)
    if (is.null(risk_cache[[k]])) risk_cache[[k]] <- pre_raster("slope_grid", res)
    if (is.null(risk_cache[[k]])) {
      tpl <- get_template(res)
      s <- tryCatch(terra::project(India_slope1, tpl, method = "max"),   # keep steepest value
                    error = function(e) terra::project(India_slope1, tpl, method = "bilinear"))
      risk_cache[[k]] <- terra::subst(s, NA, 0)
    }
    risk_cache[[k]]
  }
  
  #Scores the highest slope according to a 3 x 3 neighbourhood
  get_slope_max <- function(res) {
    k <- paste0("slopemax_", res)
    if (is.null(risk_cache[[k]])) risk_cache[[k]] <- pre_raster("slope_max", res)
    if (is.null(risk_cache[[k]])) {
      risk_cache[[k]] <- terra::subst(terra::focal(get_slope_grid(res), w = 3, fun = "max", na.rm = TRUE), NA, 0)
    }
    risk_cache[[k]]
  }
  
  get_glacier_counts <- function(res) {
    k <- paste0("glaciers_", res)
    if (is.null(risk_cache[[k]])) risk_cache[[k]] <- pre_raster("glacier_counts", res)
    if (is.null(risk_cache[[k]])) {
      gl <- st_transform(glacierview_clean, 32643)
      if (!all(st_geometry_type(gl) %in% c("POINT", "MULTIPOINT"))) {
        gl <- suppressWarnings(st_point_on_surface(gl))
      }
      risk_cache[[k]] <- terra::rasterize(terra::vect(gl), get_template(res), field = 1,
                                          fun = "sum", background = 0)
    }
    risk_cache[[k]]
  }
  
  #Glacier density is scaled here from 0 to 1
  get_kde_score <- function(res, H) {
    k <- paste0("kde_", res, "_", H)
    if (is.null(risk_cache[[k]])) risk_cache[[k]] <- pre_raster("kde", paste0(res, "_", H))
    if (is.null(risk_cache[[k]])) {
      kde  <- kde_quartic(get_glacier_counts(res), H, res)
      kmax <- terra::global(kde, "max", na.rm = TRUE)[[1]]
      sc   <- if (is.finite(kmax) && kmax > 0) kde / kmax else kde * 0
      risk_cache[[k]] <- terra::clamp(terra::subst(sc, NA, 0), lower = 0, upper = 1, values = TRUE)
    }
    risk_cache[[k]]
  }

      risk_cache[[k]] <- terra::rasterize(v, get_template(res), field = 1)
    }
    risk_cache[[k]]
  }
  
  area_km2 <- function(freq_df, v, res) {
    n <- freq_df$count[freq_df$value == v]
    if (length(n) == 0) 0 else round(n * res^2 / 1e6)
  }
  
#Glacier Exposure
  # Three 0-1 scores on a common UTM 43N grid, averaged into one risk surface:
  #  1. River proximity : In terms of IDW^2 where regions close to the river are 1 and regions further awahy are 0. 
  #  2. Glacier density : Quartic KDE, where regions falling inside the bandwith are 1 and outside are 0.
  #  3. Slope steepness : Similar approach to the above two, where the user input determines which slope is considered.
  #The following exposure classes are determined: Low (0-0.33], Moderate (0.33-0.67], High (0.67-1].
  #It is important to note that an exposure value of 0 is not plotted. 
  #The grids used here have already been pre-calculated
  observeEvent(input$run_risk, {
    D   <- as.numeric(input$river_distance)
    H   <- as.numeric(input$glacier_radius)
    lo  <- input$slope_range[1]
    hi  <- input$slope_range[2]
    res <- pick_res(D, input$risk_res)
    d0  <- res / 2
    
    if (hi <= lo) {
      showNotification("Slope ramp: the maximum must be greater than the minimum.", type = "error")
      return()
    }
    if (D <= d0) {
      showNotification(paste0("A ", res, " m grid is too coarse for a ", D,
                              " m river distance. Choose a finer grid cell size."), type = "error")
      return()
    }
    if (res > D / 2) {
      showNotification(paste0("Note: ", res, " m cells are coarse for a ", D,
                              " m distance; results will look blocky."), type = "warning")
    }
    
    out <- tryCatch({
      withProgress(message = "Running glacier exposure analysis...", value = 0, {
        incProgress(0.2, detail = "Reading the pre-calculated grids...")
        rd   <- get_river_dist(res)       # metres to the nearest river
        kde  <- get_kde_score(res, H)     # glacier density score, 0-1
        smax <- get_slope_max(res)        # steepest slope in each 3 x 3 neighbourhood
        
        incProgress(0.4, detail = "Scoring (IDW squared, KDE, slope) and classifying...")
        cls <- terra::lapp(c(rd, kde, smax), fun = function(rd, kde, smax) {
          d_eff <- pmax(rd, d0)
          river <- ifelse(rd <= D, (1 / d_eff^2 - 1 / D^2) / (1 / d0^2 - 1 / D^2), 0)
          slope <- pmin(pmax((smax - lo) / (hi - lo), 0), 1)
          r     <- (river + kde + slope) / 3
          ifelse(r > 2/3, 3, ifelse(r > 1/3, 2, ifelse(r > 0, 1, NA)))   # 1 Low, 2 Moderate, 3 High
        }, wopt = list(datatype = "INT1U"))
        
        incProgress(0.3, detail = "Preparing map...")
        list(ll = display_raster(cls), freq = terra::freq(cls))
      })
    }, error = function(e) {
      showNotification(paste("Glacier exposure error:", conditionMessage(e)), type = "error", duration = 15)
      NULL
    })
    if (is.null(out)) return()
    
    risk_result(out$ll)
    
    leafletProxy("map") %>%
      clearGroup("Glacier Exposure") %>%
      removeControl("risk_legend") %>%
      addRasterImage(out$ll, colors = risk_pal, method = "ngb", opacity = 0.75, group = "Glacier Exposure",
                     maxBytes = img_bytes, project = FALSE) %>%
      addLegend("bottomright", colors = risk_cols, labels = c("Low", "Moderate", "High"),
                title = "Glacier exposure", layerId = "risk_legend", opacity = 0.9)
    show_in_layers("Glacier Exposure", bounds = TRUE)
    
    showNotification(
      paste0("Glacier exposure complete (River IDW\u00b2 ", D, " m | Glacier KDE ", H, " m | Slope ",
             lo, "-", hi, "\u00b0 | ", res, " m cells). Area: Low ", area_km2(out$freq, 1, res),
             " km\u00b2, Moderate ", area_km2(out$freq, 2, res),
             " km\u00b2, High ", area_km2(out$freq, 3, res), " km\u00b2."),
      type = "message", duration = 10)
    rm(out); invisible(gc())
  })
  
  #Evacuation
  #Road access is determined as based on the IDW^2 where next to the road is 1 and further away is 0, with respect to the distance identified.
  #The output refers to the extent of suitability and what intervention strategy should be implemented.
  #The ranked outputs are read in a single-pass, to reduce memory.
  observeEvent(input$run_evacuation, {
    D   <- as.numeric(input$road_distance)
    res <- pick_res(D, input$evac_res)
    d0  <- res / 2
    
    if (D <= d0) {
      showNotification(paste0("A ", res, " m grid is too coarse for a ", D,
                              " m road distance. Choose a finer grid cell size."), type = "error")
      return()
    }
    if (res > D / 2) {
      showNotification(paste0("Note: ", res, " m cells are coarse for a ", D,
                              " m distance; results will look blocky."), type = "warning")
    }
    
    out <- tryCatch({
      withProgress(message = "Running evacuation analysis...", value = 0, {
        incProgress(0.2, detail = paste0("Reading the distance-to-roads grid (", res, " m cells)..."))
        rd  <- get_road_dist(res)
        msk <- get_study_mask(res)
        
        incProgress(0.4, detail = "Ranking suitability (IDW squared)...")
        cls <- terra::lapp(c(rd, msk), fun = function(rd, m) {
          ifelse(is.na(m), NA, ifelse(rd <= D / 2, 1, ifelse(rd <= D, 2, 3)))
        }, wopt = list(datatype = "INT1U"))
        
        incProgress(0.3, detail = "Preparing map...")
        list(ll = display_raster(cls), freq = terra::freq(cls))
      })
    }, error = function(e) {
      showNotification(paste("Evacuation error:", conditionMessage(e)), type = "error", duration = 15)
      NULL
    })
    if (is.null(out)) return()
    
    leafletProxy("map") %>%
      clearGroup("Evacuation Suitability") %>%
      removeControl("evac_legend") %>%
      addRasterImage(out$ll, colors = evac_pal, method = "ngb", opacity = 0.75,
                     group = "Evacuation Suitability", maxBytes = img_bytes, project = FALSE) %>%
      addLegend("bottomright", colors = evac_cols,
                labels = c(paste0("Ideal (\u2264 ", D / 2, " m)"),
                           paste0("Highly suitable (", D / 2, "\u2013", D, " m)"),
                           paste0("Not ideal (> ", D, " m) \u2013 requires intervention")),
                title = "Evacuation suitability", layerId = "evac_legend", opacity = 0.9)
    show_in_layers("Evacuation Suitability", bounds = TRUE)
    
    showNotification(
      paste0("Evacuation analysis complete (road IDW\u00b2 distance ", D, " m, ", res, " m cells). Area: Ideal ",
             area_km2(out$freq, 1, res), " km\u00b2, Highly suitable ", area_km2(out$freq, 2, res),
             " km\u00b2, Not ideal ", area_km2(out$freq, 3, res), " km\u00b2."),
      type = "message", duration = 10)
    rm(out); invisible(gc())
  })
