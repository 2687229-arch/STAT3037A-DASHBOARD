#Preprocessing

#Aim of this: to preprocess the data locally such that the workload is reduced from the app. This code essentially takes the raw data
#and cleans the data, reprojects, and clips it. This data is effectively stored into a new file called "data", which contains
#the RDS files for the app and settlement boundaries, and tiff file for Slope. The app then reads this folder without
#having to pre-process anything from scratch upon each instance of running it.

#Packages used
library(sf)
library(dplyr)
library(terra)
library(spdep)

#1 Establish folders and parameters to store data
#Working Directory
base_dir <- "Your Windows Explorer Folder"
setwd(base_dir)

#Data Folder
out_dir <- "data"                 
dir.create(out_dir, showWarnings = FALSE)

#Reduces the resolution of the data
agg_fact        <- 50

#Prevents shapefiles from being stored individually in the data folder
save_shapefiles <- FALSE 

#Coordinate System: UGS 1984 UTM Zone 43S
utm43 <- 32643                   

#2 Load the Raw Data
#Shapefiles
glacier_raw <- st_read("File Path", quiet = TRUE)
india_raw   <- st_read("File Path", quiet = TRUE)
roads_raw   <- st_read("File Path", quiet = TRUE)
rivers_raw  <- st_read("File Path", quiet = TRUE)

hp_raw <- st_read("File Path", quiet = TRUE) #Himachal Pradesh
jk_raw <- st_read("File Path", quiet = TRUE) #Jammu and Kashmir
lu_raw <- st_read("File Path",          quiet = TRUE) #Ladakh
uu_raw <- st_read("File Path",     quiet = TRUE) #Uttarakhand

#Raster
dem <- terra::rast("India_DEM.tif")

#3 Data Cleaning
#Remove the third dimension such that the data can be viewed in Leaflet and remove NA values. 
clean_vec <- function(x) {
  x |>
    sf::st_zm() |>
    dplyr::mutate(dplyr::across(where(is.numeric),   ~ dplyr::na_if(.x, -9999)),
                  dplyr::across(where(is.character), ~ dplyr::na_if(.x, "-9999")))
}

glacier_clean <- clean_vec(glacier_raw)
india_clean   <- clean_vec(india_raw)
roads_clean   <- clean_vec(roads_raw)
rivers_clean  <- clean_vec(rivers_raw)
hp_clean <- clean_vec(hp_raw); jk_clean <- clean_vec(jk_raw)
lu_clean <- clean_vec(lu_raw); uu_clean <- clean_vec(uu_raw)

#Replace -9999 No Data values with NA.
dem <- terra::subst(dem, -9999, NA)

#4 Reproject Layers to WGS 1984 UTM Zone 43S 
india_utm   <- st_transform(india_clean,   utm43)
glacier_utm <- st_transform(glacier_clean, utm43)
roads_utm   <- st_transform(roads_clean,   utm43)
rivers_utm  <- st_transform(rivers_clean,  utm43)
hp <- st_transform(hp_clean, utm43); jk <- st_transform(jk_clean, utm43)
lu <- st_transform(lu_clean, utm43); uu <- st_transform(uu_clean, utm43)
dem_utm <- terra::project(dem, paste0("EPSG:", utm43))

#Remove raw files from being saved in the data folder
rm(glacier_raw, india_raw, roads_raw, rivers_raw, hp_raw, jk_raw, lu_raw, uu_raw,
   glacier_clean, india_clean, roads_clean, rivers_clean, hp_clean, jk_clean, lu_clean, uu_clean, dem) 
invisible(gc())

#5 Encode the four regions into UTF-8 to preserve the true letters of each region, and clip accordingly
nm <- enc2utf8(as.character(india_utm$shapeName))
India_isolated <- india_utm[grepl("^Jammu|^Lad|^Him|^Uttar.{1,2}khand", nm), ]
stopifnot("Expected exactly 4 regions in the India boundary file" = nrow(India_isolated) == 4)
India_isolated <- st_make_valid(India_isolated)

#6 Clip the glaciers and roads
#Glaicers clipped to fall inside the four regions
glaciers_clip <- st_intersection(glacier_utm, India_isolated)

#Roads isolated to be National and State Highways, and clipped to the four regions
stopifnot("The roads file has no 'rdtype' column" = "rdtype" %in% names(roads_utm))
roads_hw <- roads_utm |> dplyr::filter(rdtype %in% c("National Highway", "State Highway"))

#7 Derive the slope data
slope       <- terra::terrain(dem_utm, v = "slope", unit = "degrees")
slope_small <- terra::aggregate(slope, fact = agg_fact, fun = "mean", na.rm = TRUE)
terra::writeRaster(slope_small, file.path(out_dir, "File Name.tiff"), overwrite = TRUE)

#Store the raster with the colour range
slope_rng <- as.numeric(terra::minmax(slope_small, compute = TRUE))   

rm(dem_utm, slope, slope_small)
invisible(gc())

#8 Save the cleaned shapefiles
if (save_shapefiles) {
  wr <- function(x, f) st_write(x, file.path(out_dir, f), delete_dsn = TRUE, quiet = TRUE,
                                layer_options = "ENCODING=UTF-8")
  wr(india_utm, "File Name.shp");      wr(India_isolated, "File Name.shp")
  wr(glaciers_clip, "File Name.shp"); wr(roads_hw, "File Name.shp")
  wr(rivers_utm, "File Name.shp")
  wr(hp, "File Name.shp"); wr(jk, "File Name.shp")
  wr(lu, "File Name.shp");  wr(uu, "File Name.shp")
}

#9 Construct the Data for the App
#Note, that the geometry column, and a unique identifier column, were retained for the sake of efficient processing of data within RDS files. 
#All other attribute columns were removed. 

#Grid extent established, with respect to the spatial reference system of these layers
risk_bb <- lapply(list(hp, jk, lu, uu), st_bbox)      

glacier_utm <- glaciers_clip[, character(0)]
roads_utm   <- roads_hw[, "rdtype"]
rivers_utm  <- rivers_utm[, character(0)]
india_utm   <- india_utm[, character(0)]

#Transforms the glacier, roads, rivers, and India shapefile into WGS 1984 for the sake of viewing in Leaflet.
w <- function(x) st_transform(x, 4326)
glacier_wgs <- w(glacier_utm)
roads_wgs   <- w(roads_utm)
rivers_wgs  <- w(rivers_utm)
india_wgs   <- w(india_utm)

#Roads and Rivers are simplified for visualisation. It is important to note that the full details of these features are used in analysis.
simp_lines <- function(x, tol) {
  y <- x |> st_transform(utm43) |> st_simplify(dTolerance = tol) |> st_transform(4326)
  y[!st_is_empty(y), ]
}
roads_display  <- simp_lines(roads_utm, 200)
rivers_display <- simp_lines(rivers_utm, 200)

#The four main regions are stored in one table for ease of viewing
hp <- w(hp)[, character(0)]; jk <- w(jk)[, character(0)]
lu <- w(lu)[, character(0)]; uu <- w(uu)[, character(0)]
hp$region <- "Himachal Pradesh"; jk$region <- "Jammu and Kashmir"
lu$region <- "Ladakh";           uu$region <- "Uttarakhand"

villages <- dplyr::bind_rows(hp, jk, uu, lu) |> st_make_valid()

#Here, the polygons data is ensured here to be used by Leaflet. It only represents valid shapes. 
villages <- suppressWarnings(st_collection_extract(villages, "POLYGON"))
villages <- villages[!st_is_empty(villages), ]
villages$vid <- seq_len(nrow(villages))

#The count of glaciers per village are calculated here.
villages$glacier_count <- lengths(st_intersects(villages, glacier_wgs))

#Pre-calculation of neighbours for Queen and Rook are performed here.
nb_queen <- poly2nb(villages, queen = TRUE)
nb_rook  <- poly2nb(villages, queen = FALSE)
villages_attr <- as.data.frame(st_drop_geometry(villages))[, c("region", "vid", "glacier_count")]

#The detailed layers of the four regions are read and used for the LISA analysis.
region_slug <- c("Himachal Pradesh" = "hp", "Jammu and Kashmir" = "jk", "Ladakh" = "lu", "Uttarakhand" = "uu")
for (r in names(region_slug)) {
  saveRDS(villages[villages$region == r, "vid"],
          file.path(out_dir, paste0("villages_", region_slug[[r]], ".rds")), compress = "gzip")
}

#The villages of the four main regions are stored as lightweight (at vertices at 5%) to prevent intense processing
villages_display <- if (requireNamespace("rmapshaper", quietly = TRUE)) {
  rmapshaper::ms_simplify(villages, keep = 0.05, keep_shapes = TRUE)
} else {
  villages |> st_transform(utm43) |> st_simplify(dTolerance = 300, preserveTopology = TRUE) |>
    st_transform(4326)
}
villages_display <- villages_display |> st_make_valid()
villages_display <- suppressWarnings(st_collection_extract(villages_display, "POLYGON"))
villages_display <- villages_display[!st_is_empty(villages_display), c("region", "vid", "glacier_count")]

#The four main regions have basic outlines used to draw the basemap, with the villages dissolved to increase processing speed.
#The four main regions still retain the detailed village outlines for analysis.
region_outlines <- villages |>
  st_transform(utm43) |>
  dplyr::group_by(region) |>
  dplyr::summarise(geometry = st_union(geometry), .groups = "drop") |>
  st_buffer(100) |> st_buffer(-100) |>          # closes hairline gaps between villages
  st_simplify(dTolerance = 100, preserveTopology = TRUE) |>
  st_transform(4326) |>
  st_make_valid()
region_outlines <- suppressWarnings(st_collection_extract(region_outlines, "POLYGON"))
region_outlines <- region_outlines[!st_is_empty(region_outlines), ]
stopifnot("Region outlines are empty" = nrow(region_outlines) == 4)

#Processed outputs are stored in the RDS
saveRDS(list(glacier_wgs = glacier_wgs, roads_wgs = roads_wgs, rivers_wgs = rivers_wgs,
             roads_display = roads_display, rivers_display = rivers_display,
             india_wgs = india_wgs,
             villages_attr = villages_attr, nb_queen = nb_queen, nb_rook = nb_rook,
             villages_display = villages_display, region_outlines = region_outlines,
             risk_bb = risk_bb, slope_rng = slope_rng),
        file.path(out_dir, "app_data.rds"), compress = "gzip")   # gzip reads much faster than xz

#10 Pre-calculate the grids used for the Glacier Exposure and Road Evacuation
#Since cell size is important for KDE and IDW, these cell sizes are pre-calculated here to prevent the App from
#calculating it from scratch each time.

make_template <- function(res) {      
  terra::rast(
    xmin = min(sapply(risk_bb, function(b) b[["xmin"]])),
    xmax = max(sapply(risk_bb, function(b) b[["xmax"]])),
    ymin = min(sapply(risk_bb, function(b) b[["ymin"]])),
    ymax = max(sapply(risk_bb, function(b) b[["ymax"]])),
    resolution = res, crs = paste0("EPSG:", utm43))
}

#Performs Quartic KDE
kde_quartic <- function(counts, h, res) {
  k <- ceiling(h / res)
  ax <- (-k:k) * res
  d  <- sqrt(outer(ax^2, ax^2, "+"))
  w  <- ifelse(d < h, (3 / pi) * (1 - (d / h)^2)^2 / h^2 * 1e6, 0)
  terra::focal(counts, w = w, fun = "sum", na.rm = TRUE)
}
gdal_opts <- c("COMPRESS=DEFLATE", "PREDICTOR=2")
save_grid <- function(r, f, type) {
  terra::writeRaster(r, file.path(out_dir, f), overwrite = TRUE, datatype = type, gdal = gdal_opts)
}
line_dist <- function(lines_utm, tpl) {
  r <- terra::rasterize(terra::vect(lines_utm), tpl, field = 1, touches = TRUE)
  terra::round(terra::clamp(terra::distance(r), upper = 1001, values = TRUE))
}

slope_small <- terra::rast(file.path(out_dir, "India_slope_small.tif"))
gl_pts <- if (all(st_geometry_type(glacier_utm) %in% c("POINT", "MULTIPOINT"))) glacier_utm else
  suppressWarnings(st_point_on_surface(glacier_utm))
mask_vect <- terra::vect(st_geometry(st_transform(region_outlines, utm43)))

#Grid size for Glacier Exposure lying at 500 and 250 metres
for (res in c(500, 250)) {
  cat("Pre-calculating", res, "m grids...\n")
  tpl <- make_template(res)
  save_grid(line_dist(rivers_utm, tpl), paste0("river_dist_", res, ".tif"), "INT2U")
  save_grid(line_dist(roads_utm,  tpl), paste0("road_dist_",  res, ".tif"), "INT2U")
  

#Identifies the steepest slope that one could choose, and pre-calculates
  sl <- tryCatch(terra::project(slope_small, tpl, method = "max"),
                 error = function(e) terra::project(slope_small, tpl, method = "bilinear"))
  sl <- terra::subst(sl, NA, 0)
  save_grid(sl, paste0("slope_grid_", res, ".tif"), "FLT4S")
  save_grid(terra::subst(terra::focal(sl, w = 3, fun = "max", na.rm = TRUE), NA, 0),
            paste0("slope_max_", res, ".tif"), "FLT4S")
  
#KDE and counts of glaciers per search radius, with respect to cell, is calculated here for 1000 and 2000 metres
  gcount <- terra::rasterize(terra::vect(gl_pts), tpl, field = 1, fun = "sum", background = 0)
  save_grid(gcount, paste0("glacier_counts_", res, ".tif"), "INT2U")
  for (H in c(1000, 2000)) {
    kde  <- kde_quartic(gcount, H, res)
    kmax <- terra::global(kde, "max", na.rm = TRUE)[[1]]
    ksc  <- if (is.finite(kmax) && kmax > 0) kde / kmax else kde * 0
    ksc  <- terra::clamp(terra::subst(ksc, NA, 0), lower = 0, upper = 1, values = TRUE)
    save_grid(ksc, paste0("kde_", res, "_", H, ".tif"), "FLT4S")
  }
  
  save_grid(terra::rasterize(mask_vect, tpl, field = 1), paste0("study_mask_", res, ".tif"), "INT1U")
  invisible(gc())
}

#Concatenation of the final performance of the pre-processing to provide messages to state the files are saved, and 
#the storage size of each file
cat("Done. Saved", file.path(out_dir, "app_data.rds"), "and", file.path(out_dir, "India_slope_small.tif"), "\n")
cat("app_data.rds size:", round(file.size(file.path(out_dir, "app_data.rds")) / 1e6, 1), "MB\n")
cat("Total size of the data folder:", round(sum(file.size(list.files(out_dir, full.names = TRUE))) / 1e6, 1), "MB\n")
cat("Villages:", nrow(villages), "| Glaciers:", nrow(glacier_wgs),
    "| Slope range:", round(slope_rng[1], 1), "to", round(slope_rng[2], 1), "degrees\n")
